syntax = "proto3";


option csharp_namespace = "StatisticService.Grpc.Payment";
package payment;

service InternalPaymentGrpcService {
    rpc GetRevenueStats (RevenueStatsRequest) returns (RevenueStatsReply);
    rpc GetMonthlyRevenue (YearRequest) returns (MonthlyRevenueListReply);
}

message RevenueStatsRequest {
    int32 year = 1;
    int32 month = 2;
}

message RevenueStatsReply {
    int64 totalRevenueYear = 1;
    int64 totalRevenueMonth = 2;
}

message YearRequest {
    int32 year = 1;
}

message MonthlyRevenueItem {
    int32 month = 1;
    int64 revenue = 2;
}

message MonthlyRevenueListReply {
    repeated MonthlyRevenueItem items = 1;
}
